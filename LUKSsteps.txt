PART 4: CREATE AUTO-UNLOCK SCRIPT (5 minutes)

Step 13: Create the Auto-Unlock Script

PASTE THIS ENTIRE SCRIPT:

bash
#!/bin/bash CONTAINER="/opt/secureapp.img" MOUNT_POINT="/mnt/secureapp_mount" MAP_NAME="secureapp_crypt" DEVICE_NAME="VideoAnalyzer" MAGIC_STRING="proprietary2025" log_message() {     echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" >> /var/log/secureapp-unlock.log} # Get hardware identifiers SERIAL=$(grep Serial /proc/cpuinfo 2>/dev/null | awk '{print $3}') MAC=$(ip link show 2>/dev/null | grep ether | head -1 | awk '{print $2}' | tr -d ':') CORES=$(grep -c ^processor /proc/cpuinfo 2>/dev/null) if [ -z "$SERIAL" ] || [ -z "$MAC" ] || [ -z "$CORES" ]; then     log_message "ERROR: Could not detect hardware identifiers"     exit 1 fi # Generate the key (SAME as Step 12) COMBINED="${DEVICE_NAME}:${SERIAL}:${MAC}:${CORES}:${MAGIC_STRING}" GENERATED_KEY=$(echo -n "$COMBINED" | sha256sum | cut -d' ' -f1) # Unlock log_message "Attempting unlock with hardware binding..." echo -n "$GENERATED_KEY" | cryptsetup luksOpen "$CONTAINER" "$MAP_NAME" --key-file=- 2>/dev/nullif [ -e /dev/mapper/$MAP_NAME ]; then     mount /dev/mapper/$MAP_NAME "$MOUNT_POINT" 2>/dev/null    log_message "SUCCESS: Hardware verified and container mounted"     exit 0 else     log_message "FAILURE: Hardware validation failed"     exit 1 fi
